# from https://nbviewer.org/github/SpikeAI/2023_GrimaldiPerrinet_HeterogeneousDelaySNN/blob/master/FastMotionDetection.ipynb
def make_events(movie, selectivity=1., p_target=None):
    """Encode a continuous movie into a binary spike raster (DVS-like encoder).

    A pixel emits a spike whenever the temporal derivative of the movie
    exceeds a threshold in absolute value; ON and OFF crossings are
    binarized into the same event. A one-step reset subtracts
    ``threshold * spike(t-1)`` from the current derivative, acting as a
    simple refractory mechanism: a pixel which has just spiked requires a
    stronger input to spike again at the next time step.

    When ``p_target`` is provided, the threshold is calibrated on the
    quantile of a Gaussian with the standard deviation of the derivative
    (a two-sided test), so that the mean firing rate is approximately
    ``selectivity * p_target``; otherwise the threshold is ``selectivity``
    times the standard deviation of the derivative.

    Args:
        movie: Numpy array of shape ``(N_X, N_Y, N_T)``, the continuous movie.
        selectivity: Multiplier of the calibrated baseline threshold (see above).
        p_target: Target mean firing rate per time step; if ``None``, the
            threshold is simply ``selectivity * std(difference(movie))``.

    Returns:
        np.ndarray: Binary spike raster of shape ``(N_X, N_Y, N_T)`` with
        values in ``{0, 1}``.
    """
    N_X, N_Y, N_T = movie.shape
    It = np.diff(movie, axis=-1)

    It_full_base = np.zeros((N_X, N_Y, N_T))
    It_full_base[:, :, 1:] = It.copy()

    def encode(threshold):
        It_full = It_full_base.copy()
        movie_spike = np.zeros((N_X, N_Y, N_T))
        for i_T in range(1, N_T):
            It_full[:, :, i_T] -= threshold * movie_spike[:, :, i_T-1]  # reset (refractory-like)
            movie_spike[:, :, i_T] = 1. * (np.abs(It_full[:, :, i_T]) > threshold)
        return movie_spike

    if p_target is None:
        return encode(selectivity * It.std())

      # Calibrate the threshold so the mean firing rate is close to selectivity * p_target:
      # seed with the Gaussian quantile, then refine by bisection because the
      # empirical distribution of the derivative deviates from a Gaussian.
    p_spike = min(selectivity * p_target, 1.)
    z = torch.special.ndtri(torch.tensor(1. - p_spike / 2.)).item()
    low, high = 0., 2. * z     # bracket: the rate decreases with the threshold
    while encode(high * It.std()).mean() > p_spike:
        low, high = high, 2 * high
    for _ in range(10):
        threshold = (low + high) / 2
        if encode(threshold * It.std()).mean() > p_spike:
            low = threshold
        else:
            high = threshold
    return encode(threshold * It.std())

class TravellingWaveSpikingPattern(SpikingPattern):
    """Generate frozen travelling-wave spike patterns.

    Each of the ``N_pattern`` targets is a Gabor-filtered random cloud -- a
    travelling wave moving along one of the evenly spaced directions --
    rectified to ``[-1, 1]`` and encoded into a binary spike raster with a
    DVS-like threshold-and-reset mechanism (see :func:`make_events`). The
    resulting raster has a mean activity close to ``movie_gain * opt.p_A``
    and is made periodic by copying its first ``num_delay`` frames to its end.
    """

    def __init__(self,
                 movie_gain=2.,
                 v=.2, sf_0=.15, B_sf=.05, B_theta=np.pi/18, B_V=.1,
                ):
        """Initialize the travelling-wave spike pattern generator.

        Args:
            movie_gain: Selectivity of the spike encoder; the mean firing
                rate is approximately ``movie_gain * opt.p_A``.
            v: Speed of the travelling wave, in pixels per frame.
            sf_0: Preferred spatial frequency of the Gabor envelope
                (cycles per pixel).
            B_sf: Bandwidth of the Gabor envelope in spatial frequency.
            B_theta: Angular bandwidth of the orientation tuning (radians).
            B_V: Bandwidth of the temporal-frequency tuning.
        """
        super().__init__()
        self.desc = "A Travelling wave spike pattern generator"
        self.movie_gain = movie_gain
        self.v = v
        self.sf_0 = sf_0
        self.B_sf = B_sf
        self.B_theta = B_theta
        self.B_V = B_V
        self.is_periodic = True

    def get_TW(self, angle):
        """Synthesize a travelling-wave movie along a given direction.

        A random cloud is Gabor-filtered in the Fourier domain, tuned to a
        wave travelling at speed ``self.v`` along the direction ``angle``.

        Args:
            angle: Direction of propagation (radians).

        Returns:
            np.ndarray: Rectified movie with values in ``[-1, 1]``, of shape
            ``(N_X, N_X, opt.N_time - opt.num_delay)``.
        """
        # define Fourier domain - dimensions should be even in that library to have a symmetric Fourier domain
        mc.N_X, mc.N_Y, mc.N_frame = self.N_X, self.N_X, self.opt.N_time - self.opt.num_delay # we will add hd.opt.num_delay frames of the beginning of the pattern to make it evoke a periodic pattern
        fx, fy, ft = mc.get_grids(mc.N_X, mc.N_Y, mc.N_frame)
        # define an envelope
        envelope = mc.envelope_gabor(fx, fy, ft,
            V_X=self.v*np.cos(angle), V_Y=self.v*np.sin(angle), B_V=self.B_V,
            sf_0=self.sf_0, B_sf=self.B_sf,
            theta=angle, B_theta=self.B_theta
            )

        movie = mc.random_cloud(envelope, seed=self.opt.seed)
        movie = mc.rectif(movie)
        movie = (2*movie - 1)
        return movie

    def get_TW_spike(self, angle):
        """Encode a travelling-wave movie into a binary spike raster.

        Args:
            angle: Direction of propagation (radians).

        Returns:
            torch.Tensor: Spike raster with values in ``{0, 1}``, of shape
            ``(N_X, N_X, opt.N_time - opt.num_delay)``, with mean activity
            approximately ``self.movie_gain * self.opt.p_A``.
        """
        movie = self.get_TW(angle)
        movie_spike = make_events(movie, selectivity=self.movie_gain, p_target=self.opt.p_A)
        return torch.Tensor(movie_spike)

    def init(self, opt):
        """Draw the frozen travelling-wave target patterns.

        Builds one travelling wave per direction, evenly spaced in
        ``[0, 2*pi)``, stores the spike rasters in ``self.frozen_target``
        and makes each pattern periodic by copying its first
        ``opt.num_delay`` time steps to its end.

        Args:
            opt: A :class:`~mnesis_boilerplate.Params` instance providing
                ``N_pattern``, ``N_neuron``, ``N_time``, ``num_delay``,
                ``p_A``, ``seed`` and ``device``.
        """
        self.opt = opt
        self.N_X = int(opt.N_neuron**.5)
        # Make the target by generic travelling waves in different directions
        self.frozen_target = torch.zeros((opt.N_pattern, opt.N_neuron, opt.N_time), device=opt.device)
        for j_pattern, angle in enumerate(np.linspace(0, 2*np.pi, opt.N_pattern, endpoint=False)):
            TW = self.get_TW_spike(angle=angle)
            self.frozen_target[j_pattern, :, :(-opt.num_delay)] = TW.reshape((opt.N_neuron, opt.N_time - opt.num_delay))

        # Make the target learning a period sequence by copying the first opt.num_delay time steps to the end of the sequence
        self.frozen_target[:, :, (-opt.num_delay):] = self.frozen_target[:, :, :opt.num_delay]

def objective(trial):
    new_opt = Params(**opt_dict)

    new_opt.do_deconv = trial.suggest_categorical('do_deconv', [True, False])
    new_opt.do_pinv = trial.suggest_categorical('do_pinv', [True, False])

    new_opt.learn_beta = trial.suggest_categorical('learn_beta', [True, False])
    new_opt.learn_threshold = trial.suggest_categorical('learn_threshold', [True, False])
    new_opt.reset_mechanism = trial.suggest_categorical('reset_mechanism', ['zero', 'subtract'])
    new_opt.optimizer = trial.suggest_categorical('optimizer', ['adamw', 'rmsprop', 'adadelta', 'sgd', 'adam']) # 'adagrad', 'sparseadam',

    # new_opt.surrogate_name = trial.suggest_categorical('surrogate_name', ['FastSigmoid', 'LeakySpikeOperator', 'ATan', 'Sigmoid', 'SpikeRateEscape'])
    # new_opt.loss_name = trial.suggest_categorical('loss_name', [  'SpikeF1scoreLoss', 'MSELoss', 'L1Loss'])

    new_opt.lif_beta = trial.suggest_float('lif_beta', 0, 1)
    scale = 10
    scale = 3.618
    new_opt.alpha_surrogate = trial.suggest_float('alpha_surrogate', new_opt.alpha_surrogate / scale, new_opt.alpha_surrogate * scale, log=True)
    new_opt.base_lr = trial.suggest_float('base_lr', new_opt.base_lr / scale, new_opt.base_lr * scale, log=True)
    new_opt.final_lr = trial.suggest_float('final_lr', new_opt.final_lr / scale, new_opt.final_lr * scale, log=True)
    new_opt.delta1 = trial.suggest_float('delta1', new_opt.delta1 / scale, min((new_opt.delta1 * scale, .99)), log=True)
    new_opt.delta2 = trial.suggest_float('delta2', new_opt.delta2 / scale, min((new_opt.delta2 * scale, .99)), log=True)
    new_opt.dropout = trial.suggest_float('dropout', 0, 1)
    new_opt.lif_threshold = trial.suggest_float('lif_threshold', new_opt.lif_threshold / scale, new_opt.lif_threshold * scale, log=True)

    try:
        hd = HD_SNN(new_opt, TravellingWaveSpikingPattern())

        with torch.no_grad():
            # draw causes (SMs) - hidden to the observer

            spikes_ = spikes[:, :, (hd.opt.N_pretime+hd.opt.num_delay):(hd.opt.N_time+hd.opt.N_pretime)]
            loss_val = loss_fn(spikes_, target_[:, :, hd.opt.num_delay:])

        return loss_val.item()
